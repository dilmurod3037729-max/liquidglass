cmake_minimum_required(VERSION 3.16)
project(liquidglass VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(liquidglass src/scene.cpp)
target_include_directories(liquidglass PUBLIC include)
if(MSVC)
  target_compile_options(liquidglass PRIVATE /W4)
else()
  target_compile_options(liquidglass PRIVATE -Wall -Wextra)
endif()

option(LG_BUILD_NATIVE "OpenGL backend + GLFW/ImGui namunasi (internet kerak)" OFF)
option(LG_BUILD_QT "Qt 6 backend namunasi (QOpenGLWidget; Qt 6 kerak)" OFF)

# OpenGL 3.3 chizuvchi (GLFW va Qt namunalari uchun umumiy, oynalash kutubxonasiga bog'liq emas)
if(LG_BUILD_NATIVE OR LG_BUILD_QT)
  add_library(liquidglass_gl native/gl_renderer.cpp)
  target_include_directories(liquidglass_gl PUBLIC native)
  target_link_libraries(liquidglass_gl PUBLIC liquidglass)
  if(WIN32)
    target_compile_definitions(liquidglass_gl PUBLIC NOMINMAX)
    target_link_libraries(liquidglass_gl PUBLIC opengl32)
  else()
    find_package(OpenGL REQUIRED)
    target_link_libraries(liquidglass_gl PUBLIC OpenGL::GL)
  endif()
endif()
if(LG_BUILD_NATIVE)
  add_subdirectory(native)
endif()
if(LG_BUILD_QT)
  add_subdirectory(qt)
endif()

enable_testing()
add_executable(test_scene tests/test_scene.cpp)
target_link_libraries(test_scene PRIVATE liquidglass)
add_test(NAME scene COMMAND test_scene)

add_executable(dump_golden tests/dump_golden.cpp)
target_link_libraries(dump_golden PRIVATE liquidglass)
